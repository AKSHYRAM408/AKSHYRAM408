# sentysis/core.py

class SentimentEngine:
    """
    NLP pipeline that classifies
    text sentiment at scale.
    
    Built with Scikit-learn,
    validated on real-world data,
    ATS-verified for production.
    """
    
    def analyze(self, text: str) -> dict:
        cleaned  = self.preprocess(text)
        features = self.vectorize(cleaned)
        label    = self.model.predict(features)
        return {
            "sentiment": label,
